cmake_minimum_required(VERSION 3.16)
project(browser_samp_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB RAKNET_SOURCES vendor/raknet/Source/*.cpp)
# Match upstream's supported source list, avoiding optional plugins.
set(RAKNET_NAMES BitStream CheckSum CommandParserInterface ConsoleServer DataBlockEncryptor DS_HuffmanEncodingTree GetTime InternalPacketPool NetworkTypes RakClient RakNetCommandParser RakNetworkFactory RakPeer rakserver RakSleep Rand ReliabilityLayer rijndael RPCMap SHA1 SimpleMutex SocketLayer StringCompressor StringTable PluginInterface)
set(RAKNET_SOURCES vendor/raknet/SAMPRakNet.cpp)
foreach(name IN LISTS RAKNET_NAMES)
 list(APPEND RAKNET_SOURCES vendor/raknet/Source/${name}.cpp)
endforeach()
add_library(raknet STATIC ${RAKNET_SOURCES})
target_compile_definitions(raknet PUBLIC RAKNET_LEGACY PACKETS_PER_TICK=3 __stdcall=)
target_include_directories(raknet PUBLIC vendor/raknet/Include/raknet)
target_link_libraries(raknet PUBLIC pthread)
add_executable(poc-worker worker.cpp)
target_include_directories(poc-worker PRIVATE vendor)
target_link_libraries(poc-worker PRIVATE raknet)
enable_testing()
add_executable(protocol-tests tests.cpp)
target_include_directories(protocol-tests PRIVATE vendor)
target_link_libraries(protocol-tests PRIVATE raknet)
add_test(NAME protocol-codecs COMMAND protocol-tests)
